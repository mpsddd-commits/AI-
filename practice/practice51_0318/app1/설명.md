# Spark 관련 라이브러리 (데이터 처리)
from pyspark.sql import SparkSession, Row

# DB 연결용 (현재 코드에서는 크게 활용 안됨)
from sqlalchemy import create_engine, inspect

from fastapi import FastAPI

# CSV 처리용

# 환경설정 (DB 주소, Spark 주소, jar 경로 등)
from settings import settings

import os


# FastAPI 앱 생성
app = FastAPI()

spark = None

# =========================
# 🔥 서버 시작 시 실행
# =========================
@app.on_event("startup")
def startup_event():
    global spark
    try:
        # 🔥 JDBC 드라이버 (.jar 파일) 경로
        # ex) mariadb-java-client-3.x.x.jar
        jar_path = settings.jar_path

        # Spark 세션 생성
        spark = SparkSession.builder \
            .appName("mannersmakeman") \  # Spark 애플리케이션 이름
            .master(settings.spark_url) \  # 실행 환경 (local[*] 또는 클러스터 주소)
            .config("spark.jars", jar_path) \  # 🔥 JDBC 드라이버 등록 (필수)
            .config("spark.driver.host", settings.host_ip) \  # 드라이버 IP 설정
            .config("spark.driver.bindAddress", "0.0.0.0") \  # 외부 접근 허용
            .config("spark.driver.port", "10000") \  # 드라이버 포트
            .config("spark.blockManager.port", "10001") \  # 블록 매니저 포트
            .config("spark.cores.max", "2") \  # 사용할 CPU 코어 수 제한
            .config("spark.sql.sources.jdbc.driver.kind", "mariadb") \  # JDBC 종류
            .config("spark.sql.dialect", "mysql") \  # SQL 문법 설정
            .getOrCreate()  # 기존 세션 있으면 재사용

        print("✅ Spark 세션 생성 성공!")

    except Exception as e:
        print(f"❌ Spark 세션 생성 실패: {e}")


# =========================
# 🔻 서버 종료 시 실행
# =========================
@app.on_event("shutdown")
def shutdown_event():
    # Spark 리소스 정리
    if spark:
        spark.stop()


# =========================
# 📌 기본 API (테스트용)
# =========================
@app.get("/")
def read_root():

    # Spark 초기화 여부 확인
    if not spark:
        return {"status": False, "error": "Spark session not initialized"}

    try:
        # CSV 파일 읽기 (pandas)
        df = pd.read_csv(
            settings.file_dir,
            encoding="utf-8",
            header=0,
            thousands=',',
            quotechar='"',
            skipinitialspace=True
        )

        # pandas → Spark DataFrame 변환
        spDf = spark.createDataFrame(df)

        # 일부 데이터만 조회 후 JSON 변환
        result = spDf.limit(50).toPandas().to_dict(orient="records")

        return {"status": True, "data": result}

    except Exception as e:
        return {"status": False, "error": str(e)}

# 📌 데이터 적재 API
# =========================
@app.get('/hangover')
def read():

    # 현재 파일 위치 기준으로 data 폴더 경로 생성
    current_path = os.path.dirname(os.path.abspath(__file__))
    data_path = os.path.join(current_path, "data")

    # data 폴더 내 모든 파일 목록 조회
    all_files = os.listdir(data_path)

    # 파일 하나씩 처리
    for file in all_files:
        file_path = os.path.join(data_path, file)
        print("📂 파일 경로:", file_path)

        # CSV 읽기 (pandas)
        df = pd.read_csv(
            file_path,
            encoding="utf-8",
            header=0,
            thousands=',',
            quotechar='"',
            skipinitialspace=True
        )

        # 컬럼명 공백 제거
        df.columns = df.columns.str.strip()

        # 필요한 컬럼만 선택
        cols = ['고유역번호(외부역코드)', '위도', '경도']
        df2 = df[cols].copy()

        # 컬럼명 변경 (영문)
        df2.columns = ['station_id', 'latitude', 'longitude']

        print("📊 전처리 데이터:\n", df2)

        # pandas → Spark DataFrame 변환
        sDf = spark.createDataFrame(df2)

        # =========================
        # 🔥 DB 연결 정보
        # =========================
        db_properties = {
            "user": "root",
            "password": "1234",
            "driver": "org.mariadb.jdbc.Driver",  # 🔥 JDBC 드라이버
            "sessionInitStatement": "SET SQL_MODE='ANSI_QUOTES'"
        }

        try:
            # =========================
            # 🔥 MariaDB에 데이터 저장
            # =========================
            sDf.write.jdbc(
                url=f"{settings.db_url}?useUnicode=true&characterEncoding=UTF-8&sessionVariables=sql_mode='ANSI_QUOTES'",
                table="coordinate",      # 테이블명
                mode="overwrite",        # ❗ 기존 데이터 덮어쓰기 (주의!)
                # mode="append" 사용하면 기존 데이터 유지하면서 추가됨
                properties=db_properties
            )

            print(f"✅ {file} 적재 완료!")

        except Exception as e:
            print(f"❌ 적재 실패: {e}")

        # =========================
        # 🔍 저장 확인
        # =========================
        check_df = spark.read.jdbc(
            settings.db_url,
            table="coordinate",
            properties=db_properties
        )

        print("📌 데이터 개수:", check_df.count())
        check_df.show()
